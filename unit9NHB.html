// Spell (Đã sửa lỗi không chuyển câu khi làm đúng) let sIndex = 0; function startSpell() { sIndex = 0; renderSpellWord(); } function renderSpellWord() { const inputEl = document.getElementById('spellInput'); const fbEl = document.getElementById('spellFeedback'); if (inputEl) { inputEl.value = ''; inputEl.focus(); } if (fbEl) fbEl.innerText = ''; speakSpellWord(); } function speakSpellWord() { if ('speechSynthesis' in window && dataset[sIndex]) { window.speechSynthesis.cancel(); let cleanWord = dataset[sIndex].word.replace(/[^\w\s-]/gi, ''); let u = new SpeechSynthesisUtterance(cleanWord); u.lang = 'en-US'; window.speechSynthesis.speak(u); } } function checkSpell() { let input = document.getElementById('spellInput').value.trim().toLowerCase(); let target = dataset[sIndex].word.toLowerCase().replace(/[^\w\s-]/gi, '').trim(); let fb = document.getElementById('spellFeedback'); if (input === target || input === dataset[sIndex].word.toLowerCase()) { fb.style.color = '#23b26d'; fb.innerText = ' Chính xác!'; setTimeout(() => { sIndex = (sIndex + 1) % dataset.length; renderSpellWord(); }, 800); } else { fb.style.color = '#ff495c'; fb.innerText = ` Chưa đúng! Gợi ý nghĩa: ${dataset[sIndex].meaning}`; } }
